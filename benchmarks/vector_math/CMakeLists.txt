cmake_minimum_required(VERSION 3.18)
project(IrisVectorMathProbe LANGUAGES CXX)
# Standalone research executable; these dependencies do not enter Iris builds.
set(SLEEF_INCLUDE_DIR "" CACHE PATH "SLEEF header directory")
set(SLEEF_LIBRARY "" CACHE FILEPATH "Static SLEEF library")
set(MPFR_INCLUDE_DIR "" CACHE PATH "MPFR and GMP header directory")
set(MPFR_LIBRARY "" CACHE FILEPATH "MPFR library")
set(GMP_LIBRARY "" CACHE FILEPATH "GMP library")
foreach(dependency SLEEF_INCLUDE_DIR SLEEF_LIBRARY MPFR_INCLUDE_DIR MPFR_LIBRARY GMP_LIBRARY)
  if(NOT EXISTS "${${dependency}}")
    message(FATAL_ERROR "Set ${dependency} to an existing research dependency")
  endif()
endforeach()
add_executable(iris_vector_math_probe probe.cpp)
target_compile_features(iris_vector_math_probe PRIVATE cxx_std_17)
target_include_directories(iris_vector_math_probe SYSTEM PRIVATE "${SLEEF_INCLUDE_DIR}" "${MPFR_INCLUDE_DIR}")
target_compile_definitions(iris_vector_math_probe PRIVATE SLEEF_STATIC_LIBS=1)
target_link_libraries(iris_vector_math_probe PRIVATE "${SLEEF_LIBRARY}" "${MPFR_LIBRARY}" "${GMP_LIBRARY}")
if(MSVC)
  target_compile_options(iris_vector_math_probe PRIVATE /arch:AVX2 /fp:strict /W4)
else()
  target_compile_options(iris_vector_math_probe PRIVATE -mavx2 -mfma -ffp-contract=off -fno-fast-math -Wall -Wextra)
endif()
