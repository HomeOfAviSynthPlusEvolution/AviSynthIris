cmake_minimum_required(VERSION 3.16)
project(Iris VERSION 0.0.1 LANGUAGES C CXX)
option(IRIS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(IRIS_LLVM "Build the optional LLVM ORC backend (LLVM 22 C API)" OFF)
option(IRIS_SLEEF_AVX2 "Allow AVX2/FMA vector math on supported native CPUs" ON)
option(IRIS_SLEEF "Enable optional SLEEF math precision modes" OFF)
set(IRIS_SLEEF_INCLUDE_DIR "" CACHE PATH "SLEEF include directory")
set(IRIS_SLEEF_LIBRARY "" CACHE FILEPATH "Static SLEEF library")
option(IRIS_AVS "Build the Windows AVS C API plugin and integration tests" OFF)
option(IRIS_BENCHMARKS "Build standalone performance measurements" OFF)
set(IRIS_AVS_INCLUDE_DIR "" CACHE PATH "AviSynth SDK include directory")
set(IRIS_AVS_RUNTIME "" CACHE FILEPATH "Absolute path to existing avisynth.dll")
add_library(iris STATIC
  src/api/api.cpp src/frontend/rpn.cpp src/ir/ir.cpp src/ir/math.cpp
  src/runtime/runtime.cpp src/runtime/manual_lut.cpp src/scalar/scalar.cpp)
add_library(Iris::Iris ALIAS iris)
target_include_directories(iris PUBLIC include PRIVATE src)
target_compile_features(iris PUBLIC cxx_std_17)
set_target_properties(iris PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(IRIS_LLVM)
  find_package(LLVM REQUIRED CONFIG)
  if(NOT LLVM_VERSION_MAJOR EQUAL 22)
    message(FATAL_ERROR "Iris currently targets the LLVM 22 C API")
  endif()
  target_sources(iris PRIVATE src/llvm/llvm.cpp)
  target_include_directories(iris SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
  target_link_libraries(iris PRIVATE LLVM)
  target_compile_definitions(iris PRIVATE IRIS_WITH_LLVM=1)
endif()
if(IRIS_SLEEF)
  if(NOT EXISTS "${IRIS_SLEEF_INCLUDE_DIR}/sleef.h" OR NOT EXISTS "${IRIS_SLEEF_LIBRARY}")
    message(FATAL_ERROR "IRIS_SLEEF requires SLEEF headers and a static library")
  endif()
  target_include_directories(iris SYSTEM PRIVATE "${IRIS_SLEEF_INCLUDE_DIR}")
  target_link_libraries(iris PRIVATE "${IRIS_SLEEF_LIBRARY}")
  target_compile_definitions(iris PRIVATE IRIS_WITH_SLEEF=1 SLEEF_STATIC_LIBS=1)
  if(IRIS_LLVM AND IRIS_SLEEF_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$" AND NOT MSVC)
    target_sources(iris PRIVATE src/ir/math_avx2.cpp)
    set_source_files_properties(src/ir/math_avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
    target_compile_definitions(iris PRIVATE IRIS_WITH_SLEEF_AVX2=1)
  endif()
endif()
include(cmake/DeveloperTools.cmake)
if(MSVC)
  target_compile_options(iris PRIVATE /W4 /fp:strict)
else()
  target_compile_options(iris PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()
if(IRIS_SANITIZE)
  if(MSVC)
    message(FATAL_ERROR "This sanitizer configuration requires GCC or Clang")
  endif()
  target_compile_options(iris PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(iris PUBLIC -fsanitize=address,undefined)
endif()
add_executable(iris_example examples/example.c)
set_target_properties(iris_example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_link_libraries(iris_example PRIVATE Iris::Iris)
include(CTest)
if(IRIS_BENCHMARKS)
  add_executable(iris_benchmark benchmarks/benchmark.cpp)
  target_include_directories(iris_benchmark PRIVATE src)
  target_link_libraries(iris_benchmark PRIVATE Iris::Iris)
endif()
if(IRIS_AVS)
  if(NOT WIN32 OR NOT EXISTS "${IRIS_AVS_INCLUDE_DIR}/avisynth_c.h")
    message(FATAL_ERROR "IRIS_AVS needs Windows and an existing SDK")
  endif()
  add_library(iris_avs_plugin SHARED adapters/avs/plugin.cpp adapters/avs/iris_expr_avs.cpp)
  target_include_directories(iris_avs_plugin PRIVATE adapters/avs src)
  target_include_directories(iris_avs_plugin SYSTEM PRIVATE "${IRIS_AVS_INCLUDE_DIR}")
  target_link_libraries(iris_avs_plugin PRIVATE Iris::Iris)
  set_target_properties(iris_avs_plugin PROPERTIES PREFIX "" OUTPUT_NAME "IrisExpr")
  if(IRIS_BENCHMARKS)
    add_executable(iris_avs_benchmark benchmarks/avs_benchmark.cpp)
    target_include_directories(iris_avs_benchmark PRIVATE adapters/avs)
    target_include_directories(iris_avs_benchmark SYSTEM PRIVATE "${IRIS_AVS_INCLUDE_DIR}")
    target_link_libraries(iris_avs_benchmark PRIVATE Iris::Iris)
    add_dependencies(iris_avs_benchmark iris_avs_plugin)
    if(MINGW)
      target_link_options(iris_avs_benchmark PRIVATE -municode)
    endif()
    if(IRIS_LLVM)
      target_compile_definitions(iris_avs_benchmark PRIVATE IRIS_TEST_LLVM=1)
    endif()
  endif()
endif()
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(iris_host_bridge_tests tests/host_bridge_tests.c)
  set_target_properties(iris_host_bridge_tests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
  target_link_libraries(iris_host_bridge_tests PRIVATE Iris::Iris)
  add_test(NAME iris_host_bridge_tests COMMAND iris_host_bridge_tests)
  if(IRIS_LLVM)
    add_test(NAME iris_host_bridge_llvm_tests COMMAND iris_host_bridge_tests llvm)
  endif()
  add_executable(iris_math_tests tests/math_tests.cpp)
  target_include_directories(iris_math_tests PRIVATE src)
  target_link_libraries(iris_math_tests PRIVATE Iris::Iris)
  add_test(NAME iris_math_tests COMMAND iris_math_tests)
  if(IRIS_LLVM)
    add_test(NAME iris_math_llvm_tests COMMAND iris_math_tests llvm)
  endif()
  add_executable(iris_tests tests/tests.cpp)
  target_include_directories(iris_tests PRIVATE src)
  target_link_libraries(iris_tests PRIVATE Iris::Iris Threads::Threads)
  add_test(NAME iris_tests COMMAND iris_tests)
  add_executable(iris_api_v1_tests tests/api_v1_tests.c)
  set_target_properties(iris_api_v1_tests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
  target_link_libraries(iris_api_v1_tests PRIVATE Iris::Iris)
  add_test(NAME iris_api_v1_tests COMMAND iris_api_v1_tests)
  add_executable(iris_expr_tests tests/expr_tests.cpp)
  target_link_libraries(iris_expr_tests PRIVATE Iris::Iris)
  add_test(NAME iris_expr_tests COMMAND iris_expr_tests)
  add_executable(iris_manual_lut_tests tests/manual_lut_tests.cpp)
  target_include_directories(iris_manual_lut_tests PRIVATE src)
  target_link_libraries(iris_manual_lut_tests PRIVATE Iris::Iris)
  add_test(NAME iris_manual_lut_tests COMMAND iris_manual_lut_tests)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$" OR CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
    add_executable(iris_float_environment_probe tests/float_environment_probe.cpp)
    target_link_libraries(iris_float_environment_probe PRIVATE Iris::Iris)
    if(IRIS_LLVM)
      target_compile_definitions(iris_float_environment_probe PRIVATE IRIS_TEST_LLVM=1)
    endif()
    # Observational: precision-boundary differences are an accepted contract limitation.
    add_test(NAME iris_float_environment_probe_smoke COMMAND iris_float_environment_probe)
  endif()
  if(IRIS_LLVM)
    add_test(NAME iris_llvm_tests COMMAND iris_tests llvm)
    add_test(NAME iris_api_v1_llvm_tests COMMAND iris_api_v1_tests llvm)
    add_test(NAME iris_expr_llvm_tests COMMAND iris_expr_tests llvm)
    add_test(NAME iris_manual_lut_llvm_tests COMMAND iris_manual_lut_tests llvm)
  endif()
  add_test(NAME iris_c_example COMMAND iris_example)
  if(IRIS_AVS)
    if(NOT WIN32 OR NOT EXISTS "${IRIS_AVS_INCLUDE_DIR}/avisynth_c.h" OR NOT EXISTS "${IRIS_AVS_RUNTIME}")
      message(FATAL_ERROR "IRIS_AVS needs Windows, an existing SDK and absolute runtime path")
    endif()
    add_executable(iris_avs_tests tests/avs_tests.cpp adapters/avs/iris_avs.cpp adapters/avs/iris_expr_avs.cpp)
    target_include_directories(iris_avs_tests PRIVATE adapters/avs src)
    target_include_directories(iris_avs_tests SYSTEM PRIVATE "${IRIS_AVS_INCLUDE_DIR}")
    target_link_libraries(iris_avs_tests PRIVATE Iris::Iris)
    if(MINGW)
      target_link_options(iris_avs_tests PRIVATE -municode)
    endif()
    if(IRIS_LLVM)
      target_compile_definitions(iris_avs_tests PRIVATE IRIS_TEST_LLVM=1)
    endif()
    add_test(NAME iris_avs_integration COMMAND iris_avs_tests "${IRIS_AVS_RUNTIME}")
    add_test(NAME iris_avs_plugin COMMAND iris_avs_tests "${IRIS_AVS_RUNTIME}" "$<TARGET_FILE:iris_avs_plugin>")
    add_executable(iris_expr_probe tests/expr_probe.cpp)
    target_include_directories(iris_expr_probe PRIVATE adapters/avs)
    target_include_directories(iris_expr_probe SYSTEM PRIVATE "${IRIS_AVS_INCLUDE_DIR}")
    target_link_libraries(iris_expr_probe PRIVATE Iris::Iris)
    if(MINGW)
      target_link_options(iris_expr_probe PRIVATE -municode)
    endif()
    if(IRIS_LLVM)
      target_compile_definitions(iris_expr_probe PRIVATE IRIS_TEST_LLVM=1)
    endif()
    # Observational probe: differences are recorded, not declared compatible.
    add_test(NAME iris_expr_probe_smoke COMMAND iris_expr_probe "${IRIS_AVS_RUNTIME}")
    add_executable(iris_expr_scale_probe tests/expr_scale_probe.cpp)
    target_include_directories(iris_expr_scale_probe PRIVATE adapters/avs)
    target_include_directories(iris_expr_scale_probe SYSTEM PRIVATE "${IRIS_AVS_INCLUDE_DIR}")
    target_link_libraries(iris_expr_scale_probe PRIVATE Iris::Iris)
    if(IRIS_LLVM)
      target_compile_definitions(iris_expr_scale_probe PRIVATE IRIS_TEST_LLVM=1)
    endif()
    if(MINGW)
      target_link_options(iris_expr_scale_probe PRIVATE -municode)
    endif()
    add_test(NAME iris_expr_scale_probe_smoke COMMAND iris_expr_scale_probe "${IRIS_AVS_RUNTIME}")
  endif()
  if(IRIS_SANITIZE)
    get_property(iris_all_tests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${iris_all_tests} PROPERTIES
      ENVIRONMENT "ASAN_OPTIONS=halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1")
  endif()
endif()
